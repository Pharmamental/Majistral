class RootPageView extends StatefulWidget {
  const RootPageView({super.key});

  @override
  State<RootPageView> createState() => _RootPageViewState();
}

class _RootPageViewState extends State<RootPageView> {
  final PageController _pageController = PageController();
  int _currentIndex = 0;

  void _goToPage(int index) {
    final distance = (index - _currentIndex).abs().clamp(1, 6);
    _pageController.animateToPage(
      index,
      duration: Duration(milliseconds: 500 * distance),
      curve: Curves.easeInOut,
    );
  }

  @override
  void dispose() {
    _pageController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: PageView(
        controller: _pageController,
        onPageChanged: (index) => _currentIndex = index,
        children: [
          HomePage(onColorSelected: _goToPage),
          for (final color in _colorScreenColors)
            ColorPage(color: color, onBack: () => _goToPage(0)),
        ],
      ),
    );
  }
}

class _OrderBoxColorPair {
  const _OrderBoxColorPair(this.background, this.accent);

  final Color background;
  final Color accent;
}

const List<_OrderBoxColorPair> _orderBoxColors = [
  _OrderBoxColorPair(Color(0xFFE65100), Color(0xFFFFB74D)), // koyu turuncu
  _OrderBoxColorPair(Color(0xFFB71C1C), Color(0xFFE57373)), // koyu kırmızı
  _OrderBoxColorPair(Color(0xFF0D47A1), Color(0xFF64B5F6)), // koyu mavi
  _OrderBoxColorPair(Color(0xFF1B5E20), Color(0xFF81C784)), // koyu yeşil
];

class _OrderBoxData {
  _OrderBoxData({
    required this.letter,
    required this.gold,
    required this.initialInventory,
    required this.colorPair,
    required this.waitHours,
  });

  final String letter;
  final int gold;
  final int initialInventory;
  final _OrderBoxColorPair colorPair;
  final int waitHours;

  Color get lightColor => colorPair.background;
  Color get accentColor => colorPair.accent;
}

class OrderScreen extends StatefulWidget {
  const OrderScreen({super.key});

  @override
  State<OrderScreen> createState() => _OrderScreenState();
}

class _OrderScreenState extends State<OrderScreen> {
  static const _alphabet = 'ABCÇDEFGĞHIİJKLMNOÖPRSŞTUÜVYZ';

  List<_OrderBoxData>? _boxes;

  void _placeOrder() {
    final random = Random();
    setState(() {
      _boxes = List.generate(
        10,
        (_) => _OrderBoxData(
          letter: _alphabet[random.nextInt(_alphabet.length)],
          gold: (random.nextInt(10) + 1) * 50,
          initialInventory: random.nextInt(20) + 1,
          colorPair: _orderBoxColors[random.nextInt(_orderBoxColors.length)],
          waitHours: random.nextInt(6) + 1,
        ),
      );
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: const Color(0xFFEDEDF2),
      body: SafeArea(
        child: Column(
          children: [
            Padding(
              padding: const EdgeInsets.fromLTRB(16.0, 16.0, 16.0, 0),
              child: Row(
                children: [
                  _BackButton(onPressed: () => Navigator.of(context).pop()),
                  const SizedBox(width: 12),
                  const Text(
                    'Sipariş Listesi',
                    style: TextStyle(
                      fontSize: 20,
                      fontWeight: FontWeight.w800,
                      color: Colors.black87,
                    ),
                  ),
                ],
              ),
            ),
            if (_boxes == null)
              Padding(
                padding: const EdgeInsets.only(top: 24.0),
                child: ElevatedButton.icon(
                  onPressed: _placeOrder,
                  icon: const Icon(Icons.shopping_cart_checkout_rounded),
                  label: const Text('Sipariş ver'),
                  style: ElevatedButton.styleFrom(
                    backgroundColor: _kiremitColor,
                    foregroundColor: Colors.white,
                  ),
                ),
              ),
            const SizedBox(height: 12),
            if (_boxes != null)
              Expanded(
                child: ListView.builder(
                  padding: const EdgeInsets.fromLTRB(20.0, 8.0, 20.0, 20.0),
                  itemCount: _boxes!.length,
                  itemBuilder: (context, index) {
                    return OrderBoxTile(
                      key: ValueKey(index),
                      data: _boxes![index],
                    );
                  },
                ),
              ),
          ],
        ),
      ),
    );
  }
}

class OrderBoxTile extends StatefulWidget {
  const OrderBoxTile({super.key, required this.data});

  final _OrderBoxData data;

  @override
  State<OrderBoxTile> createState() => _OrderBoxTileState();
}

class _OrderBoxTileState extends State<OrderBoxTile>
    with TickerProviderStateMixin {
  late int _inventory = widget.data.initialInventory;
  bool _isRunning = false;
  int _quantity = 1;
  late final AnimationController _controller;
  late final AnimationController _shimmerController;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      vsync: this,
      duration: const Duration(seconds: 10),
    );
    _shimmerController = AnimationController(
      vsync: this,
      duration: const Duration(seconds: 2),
    );
  }

  Future<void> _startTimer() async {
    if (_isRunning) return;
    final purchasedQuantity = _quantity;
    setState(() => _isRunning = true);
    _shimmerController.repeat();
    await _controller.forward(from: 0);
    if (!mounted) return;
    _shimmerController.stop();
    setState(() {
      _inventory += purchasedQuantity;
      _isRunning = false;
      _quantity = 1;
    });
  }

  void _openPurchaseSheet(BuildContext context) {
    var sheetQuantity = _quantity;
    showModalBottomSheet<void>(
      context: context,
      backgroundColor: widget.data.lightColor,
      shape: const RoundedRectangleBorder(
        borderRadius: BorderRadius.vertical(top: Radius.circular(16)),
      ),
      builder: (sheetContext) {
        return StatefulBuilder(
          builder: (sheetContext, setSheetState) {
            return Padding(
              padding: const EdgeInsets.all(24.0),
              child: Column(
                mainAxisSize: MainAxisSize.min,
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Center(
                    child: Container(
                      width: 40,
                      height: 4,
                      margin: const EdgeInsets.only(bottom: 16.0),
                      decoration: BoxDecoration(
                        color: Colors.white38,
                        borderRadius: BorderRadius.circular(2),
                      ),
                    ),
                  ),
                  Text(
                    '${widget.data.letter} harfinin karşılığı: ${widget.data.gold} gold',
                    style: const TextStyle(
                      fontSize: 16,
                      fontWeight: FontWeight.bold,
                      color: Colors.white,
                    ),
                  ),
                  const SizedBox(height: 8),
                  Text(
                    'Sipariş süresi : ${widget.data.waitHours} saat',
                    style: const TextStyle(fontSize: 14, color: Colors.white70),
                  ),
                  const SizedBox(height: 20),
                  Row(
                    children: [
                      GestureDetector(
                        onTap: () {
                          if (sheetQuantity > 1) {
                            setSheetState(() => sheetQuantity -= 1);
                          }
                        },
                        child: const Icon(
                          Icons.remove_circle_outline,
                          size: 28,
                          color: Colors.white,
                        ),
                      ),
                      Padding(
                        padding: const EdgeInsets.symmetric(horizontal: 16.0),
                        child: Text(
                          '$sheetQuantity',
                          style: const TextStyle(
                            fontSize: 20,
                            fontWeight: FontWeight.bold,
                            color: Colors.white,
                          ),
                        ),
                      ),
                      GestureDetector(
                        onTap: () => setSheetState(() => sheetQuantity += 1),
                        child: const Icon(
                          Icons.add_circle_outline,
                          size: 28,
                          color: Colors.white,
                        ),
                      ),
                      const Spacer(),
                      Text(
                        'Toplam: ${widget.data.gold * sheetQuantity} gold',
                        style: const TextStyle(
                          fontSize: 16,
                          fontWeight: FontWeight.bold,
                          color: Colors.white,
                        ),
                      ),
                    ],
                  ),
                  const SizedBox(height: 20),
                  SizedBox(
                    width: double.infinity,
                    child: ElevatedButton(
                      style: ElevatedButton.styleFrom(
                        backgroundColor: widget.data.accentColor,
                        foregroundColor: Colors.white,
                        shape: RoundedRectangleBorder(
                          borderRadius: BorderRadius.circular(4),
                        ),
                        padding: const EdgeInsets.symmetric(vertical: 12.0),
                      ),
                      onPressed: () {
                        _quantity = sheetQuantity;
                        Navigator.of(sheetContext).pop();
                        _startTimer();
                      },
                      child: const Text('Satın al'),
                    ),
                  ),
                ],
              ),
            );
          },
        );
      },
    );
  }

  @override
  void dispose() {
    _controller.stop();
    _shimmerController.stop();
    _controller.dispose();
    _shimmerController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Container(
      margin: const EdgeInsets.only(bottom: 14.0),
      padding: const EdgeInsets.all(14.0),
      decoration: BoxDecoration(
        color: widget.data.lightColor,
        borderRadius: BorderRadius.circular(16),
        boxShadow: [
          BoxShadow(
            color: widget.data.lightColor.withOpacity(0.45),
            blurRadius: 12,
            offset: const Offset(0, 6),
          ),
        ],
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          Row(
            children: [
              Container(
                width: 40,
                height: 40,
                alignment: Alignment.center,
                decoration: BoxDecoration(
                  color: Colors.white.withOpacity(0.18),
                  shape: BoxShape.circle,
                  border: Border.all(color: Colors.white38, width: 1.5),
                ),
                child: Text(
                  widget.data.letter,
                  style: const TextStyle(
                    fontSize: 18,
                    fontWeight: FontWeight.bold,
                    color: Colors.white,
                  ),
                ),
              ),
              const SizedBox(width: 12),
              Expanded(
                child: Text(
                  'Envanter : $_inventory',
                  style: const TextStyle(
                    fontSize: 13,
                    fontWeight: FontWeight.w600,
                    color: Colors.white70,
                  ),
                ),
              ),
              if (_isRunning)
                const SizedBox.shrink()
              else
                ElevatedButton(
                  style: ElevatedButton.styleFrom(
                    backgroundColor: widget.data.accentColor,
                    foregroundColor: Colors.white,
                    elevation: 0,
                    shape: RoundedRectangleBorder(
                      borderRadius: BorderRadius.circular(10),
                    ),
                    padding: const EdgeInsets.symmetric(
                      horizontal: 14.0,
                      vertical: 8.0,
                    ),
                  ),
                  onPressed: () => _openPurchaseSheet(context),
                  child: const Text('Satın al'),
                ),
            ],
          ),
          if (_isRunning) ...[
            const SizedBox(height: 10),
            SizedBox(height: 18, child: _buildProgressBar()),
          ],
        ],
      ),
    );
  }

  Widget _buildProgressBar() {
    return Container(
      decoration: BoxDecoration(
        color: Colors.black.withOpacity(0.15),
        borderRadius: BorderRadius.circular(9),
      ),
      child: ClipRRect(
        borderRadius: BorderRadius.circular(9),
        child: AnimatedBuilder(
          animation: Listenable.merge([_controller, _shimmerController]),
          builder: (context, child) {
            return ShaderMask(
              blendMode: BlendMode.srcATop,
              shaderCallback: (bounds) {
                final t = -0.5 + _shimmerController.value * 2.0;
                final accent = widget.data.accentColor;
                return LinearGradient(
                  colors: [
                    accent,
                    accent,
                    Colors.white.withOpacity(0.35),
                    accent,
                    accent,
                  ],
                  stops: [
                    (t - 0.25).clamp(0.0, 1.0),
                    (t - 0.12).clamp(0.0, 1.0),
                    t.clamp(0.0, 1.0),
                    (t + 0.12).clamp(0.0, 1.0),
                    (t + 0.25).clamp(0.0, 1.0),
                  ],
                  begin: const Alignment(-1.0, -1.0),
                  end: const Alignment(1.0, 1.0),
                ).createShader(bounds);
              },
              child: LinearProgressIndicator(
                value: _controller.value,
                backgroundColor: Colors.transparent,
                valueColor: AlwaysStoppedAnimation(widget.data.accentColor),
              ),
            );
          },
        ),
      ),
    );
  }
}
